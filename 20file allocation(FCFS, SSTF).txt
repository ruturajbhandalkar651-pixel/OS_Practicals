# DISK SCHEDULING
requests = [98, 183, 37, 122, 14, 124, 65, 67]
head = 53
seek_time = 0.1

# FCFS Algorithm
def fcfs(requests, head):
    movement = 0
    current = head

    for request in requests:
        movement += abs(current - request)
        current = request

    return movement


# SSTF Algorithm
def sstf(requests, head):
    pending = requests.copy()
    movement = 0
    current = head

    while pending:
        nearest = min(pending,
                      key=lambda x: abs(x - current))

        movement += abs(current - nearest)
        current = nearest
        pending.remove(nearest)

    return movement


# PERFORMANCE ANALYSIS
print("\nDISK SCHEDULING PERFORMANCE")

start = time.perf_counter()
fcfs_movement = fcfs(requests, head)
fcfs_execution = time.perf_counter() - start

start = time.perf_counter()
sstf_movement = sstf(requests, head)
sstf_execution = time.perf_counter() - start

total_requests = len(requests)

# Estimated access time
fcfs_total_time = fcfs_movement * seek_time
sstf_total_time = sstf_movement * seek_time

fcfs_average = fcfs_total_time / total_requests
sstf_average = sstf_total_time / total_requests

# Throughput
fcfs_throughput = total_requests / (fcfs_total_time / 1000)
sstf_throughput = total_requests / (sstf_total_time / 1000)

print("\nFCFS RESULTS")
print("Total Head Movement:", fcfs_movement)
print("Average Access Time:", round(fcfs_average, 2), "ms")
print("Throughput:", round(fcfs_throughput, 2), "requests/sec")
print("Execution Time:", round(fcfs_execution, 8), "seconds")

print("\nSSTF RESULTS")
print("Total Head Movement:", sstf_movement)
print("Average Access Time:", round(sstf_average, 2), "ms")
print("Throughput:", round(sstf_throughput, 2), "requests/sec")
print("Execution Time:", round(sstf_execution, 8), "seconds")


# FINAL COMPARISON
print("\nFINAL COMPARISON")

if sstf_movement < fcfs_movement:
    print("SSTF performs better than FCFS.")
else:
    print("FCFS performs better than SSTF.")

print("Total Disk Requests:", total_requests)
